import time
from pathlib import Path

current_file_path = Path().resolve()

# Helpers
# -----------------------------

def get_md_from_xy(positions, smoothing_window=200):
    """
    Compute movement direction (md) in degrees from x/y positions, with circular smoothing.
    positions: ndarray [T, 4] with columns [timestamp, x, y, hd]
    Returns: md array [T]
    """
    def circular_smooth(angles, smoothing_window=5):
        angles_rad = np.radians(angles)
        sin_vals = np.sin(angles_rad)
        cos_vals = np.cos(angles_rad)
        kernel = np.ones(smoothing_window) / smoothing_window
        smooth_sin = np.convolve(sin_vals, kernel, mode='same')
        smooth_cos = np.convolve(cos_vals, kernel, mode='same')
        smooth_angles = np.degrees(np.arctan2(smooth_sin, smooth_cos)) % 360
        return smooth_angles

    x = positions[:, 1]
    y = positions[:, 2]

    dx = x[1:] - x[:-1]
    dy = y[1:] - y[:-1]

    direction = np.degrees(np.arctan2(-dy, dx))
    direction = (direction + 360) % 360
    direction = circular_smooth(direction, smoothing_window=smoothing_window)

    # align length with positions by repeating last value
    direction = np.concatenate([direction, [direction[-1]]])
    return direction


def times_to_bin_indices(times, edges):
    """Map continuous times to indices of half-open bins defined by edges.
    Returns only indices that fall within [edges[0], edges[-1])."""
    idx = np.searchsorted(edges, times, side='right') - 1
    valid = (idx >= 0) & (idx < len(edges) - 1)
    return idx[valid]


def extract_matlab_obj(obj):
    """Recursively extract MATLAB structs into Python dictionaries and arrays."""
    if not (isinstance(obj, mat_struct) or isinstance(obj, dict)):
        return obj

    out = {}
    obj = obj.__dict__ if isinstance(obj, mat_struct) else obj

    for key, val in obj.items():
        if key in ['_fieldnames', 'type', 'time_unit', 'data_unit', 'name']:
            continue
        if isinstance(val, mat_struct) or isinstance(val, dict):
            out[key] = extract_matlab_obj(val)
        elif isinstance(val, np.ndarray):
            arr = []
            for v in val:
                arr.append(extract_matlab_obj(v))
            out[key] = np.array(arr)
        else:
            out[key] = val

    return out

def get_md_marginals_cv(spike_times, positions, cell_id_list,
                        n_md_bins=17, velocity_threshold=2, velocity_data=None,
                        smooth_window=200, cv_count=4, shuffle_md=False,
                        save_path=None):
    """
    Compute firing-rate *marginals over movement direction (MD) only* with CV splits.

    For each MD bin and each CV split, we average the per-bin firing rates across train/test time bins.

    Returns (lists of length cv_count):
        md_train_maps[cv]:  shape (n_md_bins, n_neurons)
        md_test_maps[cv]:   shape (n_md_bins, n_neurons)
        cv_masks:           list of boolean masks over base bins (train=True)
        base_time:          1D array of bin edges (len = n_bins+1)
        kept_neuron_ids:    order of neurons in the output arrays
        md_bins:            MD bin edges used
    """
    assert save_path is not None, "save_path must be provided"
    os.makedirs(save_path, exist_ok=True)

    seed = 111
    np.random.seed(seed)

    tau = 0.120  # seconds (approx 8 Hz bins)

    # Movement direction (smoothed)
    md = get_md_from_xy(positions, smoothing_window=smooth_window)
    if shuffle_md:
        np.random.shuffle(md)

    md_max = 360.0

    start_time = time.time()

    # Build time-bin edges (base_time)
    base_time = np.arange(spike_times[0, 0], spike_times[-1, 0] + tau, tau)
    n_points = len(base_time) - 1

    # Separate spikes by neuron, keeping only requested IDs
    neuron_idxs = np.array(sorted(np.unique(spike_times[:, 1]).astype(int)))
    keep = np.isin(neuron_idxs, cell_id_list)
    kept_neuron_ids = neuron_idxs[keep]

    # Vectorized frequency maps via histogram per neuron
    n_neurons = len(kept_neuron_ids)
    freq_maps = np.empty((n_points, n_neurons), dtype=float)
    for col, nid in enumerate(kept_neuron_ids):
        st = spike_times[spike_times[:, 1] == nid][:, 0]
        counts, _ = np.histogram(st, bins=base_time)
        freq_maps[:, col] = counts / tau

    # Optional velocity filtering (set low-velocity bins to NaN)
    if velocity_data is not None:
        avg_vel = np.empty(n_points)
        for i in range(n_points):
            mask = (velocity_data[:, 0] >= base_time[i]) & (velocity_data[:, 0] < base_time[i + 1])
            if np.any(mask):
                avg_vel[i] = np.nanmean(velocity_data[mask, 1])
            else:
                avg_vel[i] = np.nan
        low_vel_bins = np.where((np.isnan(avg_vel)) | (avg_vel < velocity_threshold))[0]
        if low_vel_bins.size > 0:
            print(f"Dropping {100*low_vel_bins.size/n_points:.2f}% of time bins due to low velocity")
            freq_maps[low_vel_bins, :] = np.nan

    # MD bin edges
    md_bins = np.linspace(0.0, md_max, n_md_bins + 1)

    # CV masks over time bins
    folds = np.array_split(np.arange(n_points), cv_count)
    cv_masks = []
    for cv in range(cv_count):
        mask = np.ones(n_points, dtype=bool)
        mask[folds[cv]] = False  # False => test
        cv_masks.append(mask)

    # Output arrays
    md_train_maps = [np.full((n_md_bins, n_neurons), np.nan) for _ in range(cv_count)]
    md_test_maps  = [np.full((n_md_bins, n_neurons), np.nan) for _ in range(cv_count)]

    print("Computing MD marginals...")
    # Loop over MD bins only
    for i in range(n_md_bins):
        md_mask = (md >= md_bins[i]) & (md < md_bins[i + 1])
        if not np.any(md_mask):
            continue
        # Map those position times to base_time bin indices
        md_idx = times_to_bin_indices(positions[md_mask, 0], base_time)
        if md_idx.size == 0:
            continue
        # For each CV, compute train/test means
        for cv in range(cv_count):
            mask_cv = cv_masks[cv]
            train_idx = md_idx[mask_cv[md_idx]]
            test_idx  = md_idx[~mask_cv[md_idx]]

            if train_idx.size > 0:
                md_train_maps[cv][i, :] = np.nanmean(freq_maps[train_idx, :], axis=0)
            if test_idx.size > 0:
                md_test_maps[cv][i, :] = np.nanmean(freq_maps[test_idx, :], axis=0)

    runtime = (time.time() - start_time) / 60.0
    print(f"Runtime: {runtime:.2f} minutes")

    # Save neuron ordering early for downstream alignment
    np.save(Path(save_path) / f"neuron_id_order_md_{n_md_bins}.npy", kept_neuron_ids)

    return md_train_maps, md_test_maps, cv_masks, base_time, kept_neuron_ids, md_bins

def main():
    parser = argparse.ArgumentParser(description="Compute MD-only marginal rate maps with CV from spike and position data.")
    parser.add_argument("--dataset", required=True, type=int, help="Dataset number (required).")
    parser.add_argument("--rat", required=True, type=str, help="Rat ID (required).")
    parser.add_argument("--n_md_bins", default=17, type=int, help="Number of MD bins (default: 17)")
    parser.add_argument("--smooth_window", required=True, type=int, help="Smoothing window for MD calculation (default: 200).")
    parser.add_argument("--vel_threshold", required=False, type=int, default=2, help="Velocity threshold in cm/s (default: 2).")
    parser.add_argument("--shuffle_md", required=False, action='store_true', help="Shuffle MD for control analysis.")
    parser.add_argument("--cv_count", required=False, type=int, default=4, help="Number of CV splits (default: 4)")
    args = parser.parse_args()

    # Load epochs
    epochs = loadmat(current_file_path / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/BehavEpochs.mat",
                     struct_as_record=False, squeeze_me=True)
    if args.dataset == 6:
        epoch_data = extract_matlab_obj(epochs['wakeEp'])
    else:
        epoch_data = extract_matlab_obj(epochs['wake1Ep'])

    # Load tracking -> positions [t, x, y, hd]
    tracking_data = pd.read_csv(current_file_path / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/Tracking_data.csv")
    tracking_data = tracking_data.set_axis(['timestamp', 'x', 'y', 'hd', 'pitch', 'roll'], axis=1)
    positions = tracking_data[['timestamp', 'x', 'y', 'hd']].to_numpy()
    position_mask = (positions[:, 0] > epoch_data['start']) & (positions[:, 0] < epoch_data['stop'])
    positions = positions[position_mask]

    # Normalize positions (shift so min is 0)
    positions[:, 1] += abs(np.min(positions[:, 1]))
    positions[:, 2] += abs(np.min(positions[:, 2]))

    # Load cell types and pick HD cells (to match original behavior)
    cell_types = loadmat(current_file_path / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/CellTypes.mat",
                         struct_as_record=False, squeeze_me=True)
    hd_list = extract_matlab_obj(cell_types['hd'])

    # Load spikes and build [time, neuron_id] array
    spike_data = loadmat(current_file_path / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/SpikeData.mat",
                         struct_as_record=False, squeeze_me=True)
    spike_data = extract_matlab_obj(spike_data['S'])
    arr = []
    for a in spike_data['C']:
        arr.append(a['tsd']['t'])
    full_arr = []
    for i, a in enumerate(arr):
        for aa in a:
            full_arr.append([aa, i])
    spike_times = np.array(full_arr)
    spike_times = spike_times[np.argsort(spike_times[:, 0])]
    spike_mask = (spike_times[:, 0] > epoch_data['start']) & (spike_times[:, 0] < epoch_data['stop'])
    spike_times = spike_times[spike_mask]

    # Velocity (optional)
    velocity_data = None
    vel_path = current_file_path / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/Velocity.mat"
    if os.path.exists(vel_path):
        velocity_data = loadmat(vel_path, struct_as_record=False, squeeze_me=True)
        velocity_data = extract_matlab_obj(velocity_data['vel'])
        velocity_data = np.dstack((velocity_data['t'], velocity_data['data']))[0]
        mask = (velocity_data[:, 0] > epoch_data['start']) & (velocity_data[:, 0] < epoch_data['stop'])
        velocity_data = velocity_data[mask]

    save_path = current_file_path / f"results/dusz/Dataset_{args.dataset}/{args.rat}"
    os.makedirs(save_path, exist_ok=True)

    cell_id_list = np.where(hd_list == 1)[0]  # match original: operate on HD cells
    print(len(cell_id_list))

    md_train_maps, md_test_maps, cv_masks, base_time, kept_neuron_ids, md_bins = get_md_marginals_cv(
        spike_times=spike_times,
        positions=positions,
        cell_id_list=cell_id_list,
        n_md_bins=args.n_md_bins,
        velocity_threshold=args.vel_threshold,
        velocity_data=velocity_data,
        smooth_window=args.smooth_window,
        cv_count=args.cv_count,
        shuffle_md=args.shuffle_md,
        save_path=save_path,
    )

    # Save outputs (one file per CV split)
    suffix = "_shuffled_md" if args.shuffle_md else ""

    # Bin edges & metadata once
    np.save(save_path / f"md_bin_edges_{args.n_md_bins}{suffix}.npy", md_bins)
    np.save(save_path / f"cv_base_time_edges{suffix}.npy", base_time)

    for cv in range(len(md_train_maps)):
        np.save(save_path / f"md_marginal_train_cv{cv}_{args.n_md_bins}{suffix}.npy", md_train_maps[cv])
        np.save(save_path / f"md_marginal_test_cv{cv}_{args.n_md_bins}{suffix}.npy", md_test_maps[cv])
        np.save(save_path / f"cv_mask{cv}_md_{args.n_md_bins}{suffix}.npy", cv_masks[cv])

    print("Done. Saved MD marginal maps, neuron ordering, bin edges, and CV masks.")


if __name__ == "__main__":
    main()

def extract_matlab_obj(obj):
    """Recursively extract MATLAB structs into Python dictionaries and arrays."""
    if not (isinstance(obj, mat_struct) or isinstance(obj, dict)):
        return obj
    out = {}
    obj = obj.__dict__ if isinstance(obj, mat_struct) else obj

            out[key] = np.array(arr)
        else:
            out[key] = val
    return out


def get_md_from_xy(positions, smoothing_window=200):
    def circular_smooth(angles, smoothing_window):

        cos_vals = np.cos(angles_rad)

def bin_spike_counts(spike_times, base_time):
    """Return (n_bins, n_neurons_full) spike count matrix for given bin edges."""
    n_neurons_full = int(np.max(spike_times[:, 1])) + 1
    n_bins = len(base_time) - 1
    spike_counts = np.zeros((n_bins, n_neurons_full), dtype=float)
    for n in range(n_neurons_full):
        neuron_spikes = spike_times[spike_times[:, 1] == n, 0]
        if neuron_spikes.size == 0:
            continue
        counts, _ = np.histogram(neuron_spikes, bins=base_time)
        spike_counts[:, n] = counts
    return spike_counts

def solo_bayesian_decoding(rate_map_md, spike_counts_sel, tau, md_bins, avg_vel, vel_threshold, eps=1e-10):
    """
    Solo MD decoding using Poisson likelihood.
      rate_map_md: (n_md_bins, n_neurons)
      spike_counts_sel: (n_time_bins, n_neurons) aligned with neuron order in rate_map_md
    Returns decoded_md (n_time_bins,) and list of posteriors (each (n_md_bins,)).
    """
    n_time_bins = spike_counts_sel.shape[0]
    n_md_bins, n_neurons = rate_map_md.shape
    decoded_md = np.full(n_time_bins, np.nan)
    posterior_all = []

    # Precompute log(rate*tau) and rate*tau for speed; guard NaNs
    lam = np.nan_to_num(rate_map_md * tau, nan=0.0, posinf=0.0, neginf=0.0)
    loglam = np.log(lam + eps)

    for t in range(n_time_bins):
        if np.isnan(avg_vel[t]) or avg_vel[t] < vel_threshold:
            posterior_all.append(np.full(n_md_bins, np.nan))
            continue
        k_t = spike_counts_sel[t]  # (n_neurons,)
        # logL over md bins: sum_n [ k_n * log(lam_mn) - lam_mn ]
        logL = (k_t[None, :] * loglam - lam).sum(axis=1)
        # normalize
        max_logL = np.max(logL)
        post = np.exp(logL - max_logL)
        s = np.nansum(post)
        if s <= 0:
            posterior_all.append(np.full(n_md_bins, np.nan))
            continue
        post /= s
        posterior_all.append(post)
        md_idx = int(np.nanargmax(post))
        decoded_md[t] = (md_bins[md_idx] + md_bins[md_idx + 1]) / 2.0

    return decoded_md, posterior_all

def load_md_products(save_path, n_md_bins, use_shuffled):
    """Load saved artifacts from md_marginals.py."""
    suffix = "_shuffled_md" if use_shuffled else ""
    md_edges_fp = Path(save_path) / f"md_bin_edges_{n_md_bins}{suffix}.npy"
    neuron_order_fp = Path(save_path) / f"neuron_id_order_md_{n_md_bins}.npy"  # neuron order has no shuffle suffix
    base_edges_fp = Path(save_path) / f"cv_base_time_edges{suffix}.npy"
    if not md_edges_fp.exists():
        raise FileNotFoundError(f"Missing MD bin edges at {md_edges_fp}")
    if not neuron_order_fp.exists():
        raise FileNotFoundError(f"Missing neuron order at {neuron_order_fp}")
    md_bins = np.load(md_edges_fp)
    kept_neuron_ids = np.load(neuron_order_fp)
    base_time_edges = np.load(base_edges_fp) if base_edges_fp.exists() else None
    return md_bins, kept_neuron_ids, base_time_edges


def load_md_rate_map(save_path, n_md_bins, cv_index, use_shuffled):
    """Return (rate_map_md, file_used). If cv_index<0, average across available CV splits."""
    suffix = "_shuffled_md" if use_shuffled else ""
    if cv_index >= 0:
        fp = Path(save_path) / f"md_marginal_train_cv{cv_index}_{n_md_bins}{suffix}.npy"
        if not fp.exists():
            raise FileNotFoundError(f"MD marginal train map not found: {fp}")
        return np.load(fp), str(fp)
    # average across cv folds if present
    maps = []
    for cv in range(8):  # try up to 8; break when miss
        cand = Path(save_path) / f"md_marginal_train_cv{cv}_{n_md_bins}{suffix}.npy"
        if cand.exists():
            maps.append(np.load(cand))
    if not maps:
        # fallback to test maps if only those exist
        for cv in range(8):
            cand = Path(save_path) / f"md_marginal_test_cv{cv}_{n_md_bins}{suffix}.npy"
            if cand.exists():
                maps.append(np.load(cand))
    if not maps:
        raise FileNotFoundError(f"No MD marginal maps found in {save_path} for n_md_bins={n_md_bins}")
    stacked = np.stack(maps, axis=0)
    return np.nanmean(stacked, axis=0), "avg_of_cv_maps"

def main():
    parser = argparse.ArgumentParser(description="Bayesian decoding with optional SOLO (MD-only) mode.")
    parser.add_argument("--dataset", required=True, type=int)
    parser.add_argument("--rat", required=True, type=str)
    parser.add_argument("--n_md_bins", default=17, type=int)
    parser.add_argument("--tau", default=0.120, type=float, help="Bin size in seconds; should match marginals.")
    parser.add_argument("--smooth_window", required=True, type=int)
    parser.add_argument("--vel_threshold", default=2, type=float)
    parser.add_argument("--cv_index", default=-1, type=int, help="If >=0, decode only on held-out bins of this CV split.")
    parser.add_argument("--shuffle_md", action='store_true', help="Use shuffled-md products from training script.")
    parser.add_argument("--solo", action='store_true', help="Use MD-only marginals (from md_marginals.py) for decoding.")

    args = parser.parse_args()

    cwd = Path().resolve()
    save_path = cwd / f"results/dusz/Dataset_{args.dataset}/{args.rat}"

    # -----------------
    # Load SOLO artifacts
    # -----------------
    if not args.solo:
        raise SystemExit("This script is configured for --solo MD decoding only in this version.")

    md_bins, kept_neuron_ids, base_time_edges_saved = load_md_products(save_path, args.n_md_bins, args.shuffle_md)
    rate_map_md, src_used = load_md_rate_map(save_path, args.n_md_bins, args.cv_index, args.shuffle_md)
    if rate_map_md.ndim != 2:
        raise ValueError(f"Expected MD rate map (n_md_bins, n_neurons). Got shape {rate_map_md.shape} from {src_used}")

    # -----------------
    # Load spikes & epoch
    # -----------------
    spike_data_file = cwd / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/SpikeData.mat"
    spike_data = loadmat(spike_data_file, struct_as_record=False, squeeze_me=True)
    spike_data = extract_matlab_obj(spike_data['S'])

    arr = []
    for a in spike_data['C']:
        arr.append(a['tsd']['t'])
    full_arr = []
    for i, a in enumerate(arr):
        for aa in a:
            full_arr.append([aa, i])
    spike_times = np.array(full_arr)
    spike_times = spike_times[np.argsort(spike_times[:, 0])]

    epochs_file = cwd / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/BehavEpochs.mat"
    epochs = loadmat(epochs_file, struct_as_record=False, squeeze_me=True)
    epoch_data = extract_matlab_obj(epochs['wakeEp'] if args.dataset == 6 else epochs['wake1Ep'])
    mask = (spike_times[:, 0] > epoch_data['start']) & (spike_times[:, 0] < epoch_data['stop'])
    spike_times = spike_times[mask]

    # -----------------
    # Base-time edges (align with marginals if possible)
    # -----------------
    if base_time_edges_saved is not None:
        base_time = base_time_edges_saved
        # sanity check tau
        diffs = np.diff(base_time)
        est_tau = float(np.median(diffs))
        if abs(est_tau - args.tau) > 1e-6:
            print(f"[warn] tau mismatch: marginals tau≈{est_tau:.6f}s vs arg tau={args.tau:.6f}s. Using saved edges.")
    else:
        base_time = np.arange(spike_times[0, 0], spike_times[-1, 0] + args.tau, args.tau)

    # Spike counts in full space then select neurons
    spike_counts_full = bin_spike_counts(spike_times, base_time)
    # Align to kept neuron IDs
    spike_counts = spike_counts_full[:, kept_neuron_ids]

    # -----------------
    # Load velocity & compute per-bin avg velocity
    # -----------------
    avg_vel = np.empty(len(base_time) - 1)
    avg_vel[:] = np.nan
    vel_path = cwd / f"data/dusz/Dataset_{args.dataset}/{args.rat}/Data/Velocity.mat"
    if vel_path.exists():
        velocity_data = loadmat(vel_path, struct_as_record=False, squeeze_me=True)
        velocity_data = extract_matlab_obj(velocity_data['vel'])
        velocity_data = np.dstack((velocity_data['t'], velocity_data['data']))[0]
        vmask = (velocity_data[:, 0] > epoch_data['start']) & (velocity_data[:, 0] < epoch_data['stop'])
        velocity_data = velocity_data[vmask]
        for i in range(len(base_time) - 1):
            m = (velocity_data[:, 0] >= base_time[i]) & (velocity_data[:, 0] < base_time[i + 1])
            if np.any(m):
                avg_vel[i] = np.nanmean(velocity_data[m, 1])
    else:
        print("[warn] Velocity.mat not found; decoding will skip bins due to NaN velocities. Set --vel_threshold 0 to decode anyway.")
        avg_vel[:] = np.nan  # user can set threshold 0 to avoid skipping

    # -----------------
    # If CV specified, restrict decoding to test bins only
    # -----------------
    time_centers = (base_time[:-1] + base_time[1:]) / 2.0
    if args.cv_index >= 0:
        cv_mask_fp = Path(save_path) / f"cv_mask{args.cv_index}_md_{args.n_md_bins}{('_shuffled_md' if args.shuffle_md else '')}.npy"
        if not cv_mask_fp.exists():
            # also support legacy name without _md_
            alt = Path(save_path) / f"cv_mask{args.cv_index}_{args.n_md_bins}_{args.n_md_bins}{('_shuffled_md' if args.shuffle_md else '')}.npy"
            cv_mask_fp = alt if alt.exists() else cv_mask_fp
        cv_mask = np.load(cv_mask_fp)
        test_idx = np.where(~cv_mask)[0]
        spike_counts = spike_counts[test_idx, :]
        avg_vel = avg_vel[test_idx]
        time_centers = time_centers[test_idx]

    # -----------------
    # Decode (SOLO)
    # -----------------
    print("Starting SOLO MD Bayesian decoding from marginals...")
    t0 = time.time()
    decoded_md, posterior_all = solo_bayesian_decoding(rate_map_md, spike_counts, args.tau, md_bins, avg_vel, args.vel_threshold)
    dt = time.time() - t0
    print(f"Decoding completed in {dt:.2f} s")

    # -----------------
    # Save outputs
    # -----------------
    suffix = "_shuffled_md" if args.shuffle_md else ""
    out_base = Path(save_path)
    np.save(out_base / f"decoded_md_{args.n_md_bins}{suffix}_solo.npy" if args.cv_index < 0 else out_base / f"decoded_md_{args.n_md_bins}_cv{args.cv_index}{suffix}_solo.npy", decoded_md)
    # store posteriors as ragged object if needed
    try:
        np.save(out_base / f"posterior_{args.n_md_bins}{suffix}_solo.npy" if args.cv_index < 0 else out_base / f"posterior_{args.n_md_bins}_cv{args.cv_index}{suffix}_solo.npy", np.array(posterior_all, dtype=object))
    except Exception:
        # fallback to large 2D if all posteriors valid
        valid = [p for p in posterior_all if p is not None and not np.any(np.isnan(p))]
        if valid:
            P = np.vstack(valid)
            np.save(out_base / f"posterior_{args.n_md_bins}{suffix}_solo_valid.npy", P)

    np.save(out_base / f"time_centers{'' if args.cv_index < 0 else f'_cv{args.cv_index}'}{suffix}_solo.npy", time_centers)
    print(f"Saved decoded outputs to {out_base}")


if __name__ == "__main__":
    main()

def compute_velocity(locations: np.ndarray, base_time: np.ndarray) -> np.ndarray:
    """Compute mean velocity (units per second) within each time bin.
    locations: array columns [time, x, y, hd_deg]
    base_time: 1D array of bin edges
    Returns: (len(base_time)-1,) array of avg velocity per bin
    """
    if locations.shape[0] < 4:
        raise ValueError("locations must have >= 4 rows to estimate time step.")
    time_step = float(locations[3, 0] - locations[2, 0])
    steps = np.diff(locations[:, 1:3], axis=0)
    steps = np.vstack((steps, steps[-1]))
    speed_samples = np.linalg.norm(steps, axis=1) / max(time_step, 1e-9)
    v = np.zeros(len(base_time) - 1, dtype=float)
    for t in range(len(base_time) - 1):
        idx = np.where((locations[:, 0] >= base_time[t]) & (locations[:, 0] < base_time[t + 1]))[0]
        v[t] = 0.0 if idx.size == 0 else float(np.mean(speed_samples[idx]))
    return v


def compute_velocity_series(locations: np.ndarray) -> np.ndarray:
    """Instantaneous speed series (per sample) as columns [time, speed]."""
    if locations.shape[0] < 4:
        raise ValueError("locations must have >= 4 rows to estimate time step.")
    time_step = float(locations[3, 0] - locations[2, 0])
    steps = np.diff(locations[:, 1:3], axis=0)
    steps = np.vstack((steps, steps[-1]))
    speed_samples = np.linalg.norm(steps, axis=1) / max(time_step, 1e-9)
    return np.column_stack((locations[:, 0], speed_samples))


def times_to_bin_indices(times: np.ndarray, edges: np.ndarray) -> np.ndarray:
    """Map arbitrary timestamps to base_time bin indices (left-closed, right-open)."""
    idx = np.searchsorted(edges, times, side='right') - 1
    valid = (idx >= 0) & (idx < len(edges) - 1)
    return idx[valid]


def _gaussian_filter_hd(rate_curve: np.ndarray, smooth_sigma: Optional[float]) -> np.ndarray:
    """NaN-aware circular 1D Gaussian smoothing across head-direction bins.
    rate_curve: (H, N) or (H,) array; smoothing is along the first axis (HD).
    """
    if smooth_sigma is None or smooth_sigma <= 0:
        return rate_curve
    V = rate_curve.copy()
    isnan = np.isnan(V)
    V[isnan] = 0.0
    U = np.ones_like(rate_curve, dtype=float)
    U[isnan] = 0.0
    VV = gaussian_filter1d(V, sigma=smooth_sigma, axis=0, mode="wrap")
    UU = gaussian_filter1d(U, sigma=smooth_sigma, axis=0, mode="wrap")
    with np.errstate(invalid='ignore', divide='ignore'):
        out = VV / UU
        out[UU <= 0] = np.nan
    return out


def _gaussian_filter_xy(rate_map: np.ndarray, smooth_sigma_xy: Optional[Tuple[float, float]]) -> np.ndarray:
    """NaN-aware 2D Gaussian smoothing over (Y, X, N) or (Y, X)."""
    if smooth_sigma_xy is None:
        return rate_map
    sy, sx = smooth_sigma_xy
    if (sy is None or sy <= 0) and (sx is None or sx <= 0):
        return rate_map
    V = rate_map.copy()
    isnan = np.isnan(V)
    V[isnan] = 0.0
    U = np.ones_like(rate_map, dtype=float)
    U[isnan] = 0.0
    if rate_map.ndim == 2:
        VV = gaussian_filter(V, sigma=(sy, sx), mode="nearest")
        UU = gaussian_filter(U, sigma=(sy, sx), mode="nearest")
    else:
        VV = np.empty_like(V)
        UU = np.empty_like(U)
        for n in range(V.shape[-1]):
            VV[..., n] = gaussian_filter(V[..., n], sigma=(sy, sx), mode="nearest")
            UU[..., n] = gaussian_filter(U[..., n], sigma=(sy, sx), mode="nearest")
    with np.errstate(invalid='ignore', divide='ignore'):
        out = VV / UU
        out[UU <= 0] = np.nan
    return out


def _select_time_indices_in_hd_bin(
    locations: np.ndarray,
    base_time: np.ndarray,
    hd_l: Tuple[float, float],
    fast_bins: Optional[np.ndarray],
    timebin_tol: float = 1e-4,
) -> np.ndarray:
    """Return array of *time-bin indices* whose samples fall in the given HD bin."""
    l0, l1 = hd_l
    in_hd = np.where((locations[:, 3] >= l0) & (locations[:, 3] < l1))[0]
    if in_hd.size == 0:
        return np.array([], dtype=int)
    loc_times = locations[in_hd, 0]
    loc_idx = times_to_bin_indices(loc_times, base_time)
    if fast_bins is not None and fast_bins.size:
        mask = np.any(np.isclose(base_time[loc_idx, None], fast_bins, atol=timebin_tol), axis=1)
        loc_idx = loc_idx[mask]
    T = len(base_time) - 1
    return loc_idx[(loc_idx >= 0) & (loc_idx < T)]


def _select_time_indices_in_xy_bin(
    locations: np.ndarray,
    base_time: np.ndarray,
    x_win: Tuple[float, float],
    y_win: Tuple[float, float],
    fast_bins: Optional[np.ndarray],
    timebin_tol: float = 1e-4,
) -> np.ndarray:
    """Return array of *time-bin indices* whose samples fall in the given XY bin."""
    x0, x1 = x_win
    y0, y1 = y_win
    in_xy = np.where(
        (locations[:, 1] >= x0) & (locations[:, 1] < x1) &
        (locations[:, 2] >= y0) & (locations[:, 2] < y1)
    )[0]
    if in_xy.size == 0:
        return np.array([], dtype=int)
    loc_times = locations[in_xy, 0]
    loc_idx = times_to_bin_indices(loc_times, base_time)
    if fast_bins is not None and fast_bins.size:
        mask = np.any(np.isclose(base_time[loc_idx, None], fast_bins, atol=timebin_tol), axis=1)
        loc_idx = loc_idx[mask]
    T = len(base_time) - 1
    return loc_idx[(loc_idx >= 0) & (loc_idx < T)]


# ---------------------- Error analysis helpers --------------------- #

def _unwrap_deg(angle_deg: np.ndarray) -> np.ndarray:
    rad = np.deg2rad(angle_deg)
    rad_unwrapped = np.unwrap(rad, discont=np.pi)
    return np.rad2deg(rad_unwrapped)


def interp_circular_deg(t_src: np.ndarray, t_tgt: np.ndarray, angles_deg: np.ndarray) -> np.ndarray:
    order = np.argsort(t_src)
    t_src = t_src[order]
    angles_deg = angles_deg[order]
    ang_unwrapped = _unwrap_deg(angles_deg)
    ang_interp_unwrapped = np.interp(t_tgt, t_src, ang_unwrapped)
    ang_interp = np.mod(ang_interp_unwrapped, 360.0)
    return ang_interp


def angular_distance_deg(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    d = (a - b + 180.0) % 360.0 - 180.0
    return d


def _interp_linear_series(t_src: np.ndarray, t_tgt: np.ndarray, v_src: np.ndarray) -> np.ndarray:
    order = np.argsort(t_src)
    return np.interp(t_tgt, t_src[order], v_src[order])


def compute_ground_truth_hd(positions: np.ndarray, time_centers: np.ndarray,
                            velocity_series: np.ndarray, velocity_filter: float,
                            window: float = 0.06) -> np.ndarray:
    """Interpolate true HD at decoding centers and mask by mean speed in ±window."""
    timestamps = positions[:, 0]
    hd = positions[:, 3]  # degrees
    hd_interp = interp_circular_deg(timestamps, time_centers, hd)
    v_times = velocity_series[:, 0]
    v_vals = velocity_series[:, 1]
    velocity_interp = np.full(len(time_centers), np.nan)
    for i, tc in enumerate(time_centers):
        mask = (v_times >= tc - window) & (v_times <= tc + window)
        if np.any(mask):
            velocity_interp[i] = float(np.nanmean(v_vals[mask]))
    valid_velocity = (velocity_interp >= velocity_filter) & (~np.isnan(velocity_interp))
    ground_hd_filtered = np.where(valid_velocity, hd_interp, np.nan)
    return ground_hd_filtered


def evaluate_errors_hd(decoded_hd: np.ndarray, time_centers: np.ndarray,
                        positions: np.ndarray, velocity_series: np.ndarray,
                        vel_threshold: float) -> Dict[str, Any]:
    gt = compute_ground_truth_hd(positions, time_centers, velocity_series, vel_threshold)
    hd_err_abs = angular_distance_deg(decoded_hd, gt)
    total = int(hd_err_abs.size)
    valid_mask = ~np.isnan(hd_err_abs)
    valid = int(np.sum(valid_mask))
    stats = {
        'hd_mean_abs_error_deg': float(np.nanmean(np.abs(hd_err_abs))),
        'hd_median_abs_error_deg': float(np.nanmedian(np.abs(hd_err_abs))),
        'hd_std_abs_error_deg': float(np.nanstd(hd_err_abs)),
        'total_samples': total,
        'valid_samples': valid,
        'success_rate_pct': float(100.0 * valid / max(total, 1)),
    }
    return {'stats': stats, 'errors_deg': hd_err_abs}


def compute_ground_truth_xy(positions: np.ndarray, time_centers: np.ndarray,
                            velocity_series: np.ndarray, velocity_filter: float,
                            window: float = 0.06) -> np.ndarray:
    """Interpolate true XY at decoding centers and mask by mean speed in ±window."""
    t = positions[:, 0]
    x = positions[:, 1]
    y = positions[:, 2]
    x_interp = _interp_linear_series(t, time_centers, x)
    y_interp = _interp_linear_series(t, time_centers, y)
    v_times = velocity_series[:, 0]
    v_vals = velocity_series[:, 1]
    velocity_interp = np.full(len(time_centers), np.nan)
    for i, tc in enumerate(time_centers):
        mask = (v_times >= tc - window) & (v_times <= tc + window)
        if np.any(mask):
            velocity_interp[i] = float(np.nanmean(v_vals[mask]))
    valid = (velocity_interp >= velocity_filter) & (~np.isnan(velocity_interp))
    gt_xy = np.column_stack([x_interp, y_interp])
    gt_xy[~valid, :] = np.nan
    return gt_xy


def evaluate_errors_xy(decoded_xy: np.ndarray, time_centers: np.ndarray,
                        positions: np.ndarray, velocity_series: np.ndarray,
                        vel_threshold: float) -> Dict[str, Any]:
    
    gt_xy = compute_ground_truth_xy(positions, time_centers, velocity_series, vel_threshold)

    diffs = decoded_xy - gt_xy  # (T,2)
    dists = np.sqrt(np.nansum(diffs**2, axis=1))  # Euclidean
    total = int(dists.size)
    valid_mask = ~np.isnan(dists) # not sure that this ever really catches anything actually
    valid = int(np.sum(valid_mask))
    stats = {
        'xy_mean_abs_error': float(np.nanmean(dists)),
        'xy_median_abs_error': float(np.nanmedian(dists)),
        'xy_std_abs_error': float(np.nanstd(dists)),
        'total_samples': total,
        'valid_samples': valid,
        'success_rate_pct': float(100.0 * valid / max(total, 1)),
    }
    return {'stats': stats, 'errors': dists}

def extract_matlab_obj(obj):
    """Recursively extract MATLAB structs into Python dictionaries/arrays."""
    from scipy.io.matlab._mio5_params import mat_struct  # local import to avoid hard dep if unused
    if not (isinstance(obj, mat_struct) or isinstance(obj, dict)):
        return obj
    out: Dict[str, Any] = {}
    obj = obj.__dict__ if isinstance(obj, mat_struct) else obj
    for key, val in obj.items():
        if key in ['_fieldnames', 'type', 'time_unit', 'data_unit', 'name']:
            continue
        if isinstance(val, mat_struct) or isinstance(val, dict):
            out[key] = extract_matlab_obj(val)
        elif isinstance(val, np.ndarray):
            arr = []
            for v in val:
                arr.append(extract_matlab_obj(v))
            out[key] = np.array(arr)
        else:
            out[key] = val
    return out


def load_dusz_data(base_path: Path, dataset: int, rat: str) -> Tuple[np.ndarray, np.ndarray, Dict[str, np.ndarray], Optional[np.ndarray]]:
    """Load Dusz dataset files for a given dataset/rat.
    Returns: spike_times (M,2), positions (P,4), cell_types dict, velocity_series (Q,2 or None)
    """
    dpath = base_path / f"Dataset_{dataset}" / rat / "Data"

    # Epochs
    epochs = loadmat(dpath / 'BehavEpochs.mat', struct_as_record=False, squeeze_me=True)
    if dataset == 6:
        epoch_data = extract_matlab_obj(epochs['wakeEp'])
    else:
        epoch_data = extract_matlab_obj(epochs['wake1Ep'])

    # Tracking
    tracking = pd.read_csv(dpath / 'Tracking_data.csv')
    tracking = tracking.set_axis(['timestamp', 'x', 'y', 'hd', 'pitch', 'roll'], axis=1)
    positions = tracking[['timestamp', 'x', 'y', 'hd']].to_numpy()
    mask_pos = (positions[:, 0] > epoch_data['start']) & (positions[:, 0] < epoch_data['stop'])
    positions = positions[mask_pos]
    # Normalize positions to positive origin (as in user's script)
    positions[:, 1] += abs(float(np.nanmin(positions[:, 1])))
    positions[:, 2] += abs(float(np.nanmin(positions[:, 2])))

    # Cell types
    cell_types_raw = loadmat(dpath / 'CellTypes.mat', struct_as_record=False, squeeze_me=True)
    cell_types = extract_matlab_obj(cell_types_raw['CellTypes'] if 'CellTypes' in cell_types_raw else cell_types_raw)
    # Fallback keys
    hd_list = np.asarray(cell_types.get('hd', cell_types_raw.get('hd'))).astype(int).ravel()
    ex_list = np.asarray(cell_types.get('ex', cell_types_raw.get('ex'))).astype(int).ravel()

    # Spikes
    spike_raw = loadmat(dpath / 'SpikeData.mat', struct_as_record=False, squeeze_me=True)
    S = extract_matlab_obj(spike_raw['S'] if 'S' in spike_raw else spike_raw)
    arr = []
    for i, cell in enumerate(S['C']):
        ts = np.asarray(cell['tsd']['t']).ravel()
        if ts.size:
            arr.append(np.column_stack([ts, np.full(ts.size, i, dtype=int)]))
    spike_times = np.vstack(arr) if arr else np.empty((0, 2))
    spike_times = spike_times[np.argsort(spike_times[:, 0])]
    mask_spk = (spike_times[:, 0] > epoch_data['start']) & (spike_times[:, 0] < epoch_data['stop'])
    spike_times = spike_times[mask_spk]

    # Velocity (optional file), else derive from positions
    vel_path = dpath / 'Velocity.mat'
    velocity_series = None
    if vel_path.exists():
        vel_raw = loadmat(vel_path, struct_as_record=False, squeeze_me=True)
        vel = extract_matlab_obj(vel_raw['vel'] if 'vel' in vel_raw else vel_raw)
        velocity_series = np.dstack((vel['t'], vel['data']))[0]
        mask_v = (velocity_series[:, 0] > epoch_data['start']) & (velocity_series[:, 0] < epoch_data['stop'])
        velocity_series = velocity_series[mask_v]

    return spike_times, positions, {'hd': hd_list, 'ex': ex_list}, velocity_series


def extract_excitatory_ids(cell_types: Dict[str, np.ndarray]) -> np.ndarray:
    excit = np.asarray(cell_types['hd']).astype(int).ravel()
    return np.where(excit == 1)[0].astype(int)


def build_spike_rates(spike_times: np.ndarray, base_time: np.ndarray, tau: float, neuron_ids: np.ndarray) -> np.ndarray:
    """Vectorized per-neuron per-bin firing rates using numpy.histogram.
    Returns: spike_rates of shape (T, N_neurons)
    """
    T = len(base_time) - 1
    N = len(neuron_ids)
    rates = np.zeros((T, N), dtype=float)
    if spike_times.size == 0 or N == 0:
        return rates
    for n_idx, neuron_id in enumerate(neuron_ids):
        neuron_spk = spike_times[spike_times[:, 1] == neuron_id, 0]
        if neuron_spk.size:
            counts, _ = np.histogram(neuron_spk, bins=base_time)
            rates[:, n_idx] = counts.astype(float) / max(tau, 1e-9)
    return rates


# ------------------ HD tuning curves (marginal over XY) ------------- #

def get_hd_rate_maps_cv(
    spike_rates: np.ndarray,
    locations: np.ndarray,
    base_time: np.ndarray,
    n_hd_bins: int = 60,
    velocity_threshold: float = 5.0,
    filter_vel: bool = True,
    cv_count: int = 4,
    seed: int = 777,
):
    rng = np.random.RandomState(seed)
    v = compute_velocity(locations, base_time)
    fast_idx = np.where(v >= velocity_threshold)[0]
    fast_bins = base_time[fast_idx] if filter_vel else None
    print(
        f"Filtering {100.0 - (len(fast_idx) / max(len(base_time)-1,1)) * 100.0:.1f}% of bins due to low velocity"
    )

    hd_edges = np.linspace(0.0, 360.0 + 1e-6, n_hd_bins + 1)
    T, N = spike_rates.shape
    assert T == len(base_time) - 1, "spike_rates must match len(base_time)-1"

    folds = np.array_split(np.arange(T), cv_count)
    cv_masks: List[np.ndarray] = []
    for cv in range(cv_count):
        mask = np.ones(T, dtype=bool)
        mask[folds[cv]] = False
        cv_masks.append(mask)

    shape_map = (n_hd_bins, N)
    cv_rate_maps = [np.full(shape_map, np.nan) for _ in range(cv_count)]
    occupancy_maps = [np.zeros((n_hd_bins,), dtype=float) for _ in range(cv_count)]

    print("Creating cross-validation HD tuning curves…")
    for l in tqdm(range(n_hd_bins), desc="HD bins"):
        hd_win = (hd_edges[l], hd_edges[l + 1])
        idx = _select_time_indices_in_hd_bin(locations, base_time, hd_win, fast_bins)
        if idx.size == 0:
            continue
        for cv in range(cv_count):
            occupancy_maps[cv][l] = float(idx.size)
            mask_cv = cv_masks[cv]
            valid = idx[idx < mask_cv.size]
            cv_idx = valid[mask_cv[valid]]
            if cv_idx.size:
                cv_rate_maps[cv][l, :] = np.nanmean(spike_rates[cv_idx, :], axis=0)
    return (
        cv_rate_maps,
        occupancy_maps,
        cv_masks,
        hd_edges,
    )


def get_hd_rate_map_all(
    spike_rates: np.ndarray,
    locations: np.ndarray,
    base_time: np.ndarray,
    n_hd_bins: int = 60,
    velocity_threshold: float = 5.0,
    filter_vel: bool = True,
):
    v = compute_velocity(locations, base_time)
    fast_idx = np.where(v >= velocity_threshold)[0]
    fast_bins = base_time[fast_idx] if filter_vel else None
    hd_edges = np.linspace(0.0, 360.0 + 1e-6, n_hd_bins + 1)
    T, N = spike_rates.shape
    rate_map = np.full((n_hd_bins, N), np.nan, dtype=float)
    occupancy = np.zeros((n_hd_bins,), dtype=float)
    print("Creating all-data HD tuning curve…")
    for l in tqdm(range(n_hd_bins), desc="HD bins"):
        hd_win = (hd_edges[l], hd_edges[l + 1])
        idx = _select_time_indices_in_hd_bin(locations, base_time, hd_win, fast_bins)
        if idx.size:
            occupancy[l] = float(idx.size)
            rate_map[l, :] = np.nanmean(spike_rates[idx, :], axis=0)
    return rate_map, occupancy, hd_edges


# ------------------ XY spatial rate maps --------------------- #

def _get_xy_edges(locations: np.ndarray, n_x_bins: int, n_y_bins: int, arena_pad: float = 0.0):
    xmin, xmax = float(np.nanmin(locations[:, 1])), float(np.nanmax(locations[:, 1]))
    ymin, ymax = float(np.nanmin(locations[:, 2])), float(np.nanmax(locations[:, 2]))
    xmin -= arena_pad; xmax += arena_pad
    ymin -= arena_pad; ymax += arena_pad
    x_edges = np.linspace(xmin, xmax + 1e-6, n_x_bins + 1)
    y_edges = np.linspace(ymin, ymax + 1e-6, n_y_bins + 1)
    return x_edges, y_edges


def get_xy_rate_map_all(
    spike_rates: np.ndarray,
    locations: np.ndarray,
    base_time: np.ndarray,
    n_x_bins: int,
    n_y_bins: int,
    velocity_threshold: float = 5.0,
    filter_vel: bool = True,
    arena_pad: float = 0.0,
):
    """Build a single (all-data) XY tuning map (Y, X, N), plus occupancy (Y, X)."""
    v = compute_velocity(locations, base_time)
    fast_idx = np.where(v >= velocity_threshold)[0]
    fast_bins = base_time[fast_idx] if filter_vel else None
    x_edges, y_edges = _get_xy_edges(locations, n_x_bins, n_y_bins, arena_pad)
    T, N = spike_rates.shape
    rate_map = np.full((n_y_bins, n_x_bins, N), np.nan, dtype=float)
    occupancy = np.zeros((n_y_bins, n_x_bins), dtype=float)
    print("Creating all-data XY tuning map…")
    for iy in tqdm(range(n_y_bins), desc="Y bins"):
        y_win = (y_edges[iy], y_edges[iy+1])
        for ix in range(n_x_bins):
            x_win = (x_edges[ix], x_edges[ix+1])
            idx = _select_time_indices_in_xy_bin(locations, base_time, x_win, y_win, fast_bins)
            if idx.size:
                occupancy[iy, ix] = float(idx.size)
                rate_map[iy, ix, :] = np.nanmean(spike_rates[idx, :], axis=0)
    return rate_map, occupancy, (x_edges, y_edges)


def get_xy_rate_maps_cv(
    spike_rates: np.ndarray,
    locations: np.ndarray,
    base_time: np.ndarray,
    n_x_bins: int,
    n_y_bins: int,
    velocity_threshold: float = 5.0,
    filter_vel: bool = True,
    cv_count: int = 4,
    seed: int = 777,
    arena_pad: float = 0.0,
):
    rng = np.random.RandomState(seed)
    v = compute_velocity(locations, base_time)
    fast_idx = np.where(v >= velocity_threshold)[0]
    fast_bins = base_time[fast_idx] if filter_vel else None
    print(
        f"Filtering {100.0 - (len(fast_idx) / max(len(base_time)-1,1)) * 100.0:.1f}% of bins due to low velocity"
    )

    x_edges, y_edges = _get_xy_edges(locations, n_x_bins, n_y_bins, arena_pad)
    T, N = spike_rates.shape

    folds = np.array_split(np.arange(T), cv_count)
    cv_masks: List[np.ndarray] = []
    for cv in range(cv_count):
        mask = np.ones(T, dtype=bool)
        mask[folds[cv]] = False
        cv_masks.append(mask)

    shape_map = (n_y_bins, n_x_bins, N)
    cv_rate_maps = [np.full(shape_map, np.nan) for _ in range(cv_count)]
    occupancy_maps = [np.zeros((n_y_bins, n_x_bins), dtype=float) for _ in range(cv_count)]

    print("Creating cross-validation XY tuning maps…")
    for iy in tqdm(range(n_y_bins), desc="Y bins"):
        y_win = (y_edges[iy], y_edges[iy+1])
        for ix in range(n_x_bins):
            x_win = (x_edges[ix], x_edges[ix+1])
            idx = _select_time_indices_in_xy_bin(locations, base_time, x_win, y_win, fast_bins)
            if idx.size == 0:
                continue
            for cv in range(cv_count):
                occupancy_maps[cv][iy, ix] = float(idx.size)
                mask_cv = cv_masks[cv]
                valid = idx[idx < mask_cv.size]
                cv_idx = valid[mask_cv[valid]]
                if cv_idx.size:
                    cv_rate_maps[cv][iy, ix, :] = np.nanmean(spike_rates[cv_idx, :], axis=0)
    return (
        cv_rate_maps,
        occupancy_maps,
        cv_masks,
        (x_edges, y_edges),
    )


# ----------------------------- Decoding ----------------------------- #

def hd_bayesian_decoding(
    rate_map_hd: np.ndarray,  # (H, N)
    spike_counts: np.ndarray,  # (T, N)
    tau: float,
    hd_bins: np.ndarray,       # (H+1,)
    avg_vel: np.ndarray,       # (T,)
    vel_threshold: float,
    eps: float = 1e-10,
):
    """Head-direction Bayesian decoding with Poisson likelihood."""
    T = spike_counts.shape[0]
    H, N = rate_map_hd.shape
    decoded_hd = np.full(T, np.nan)
    posterior_all: List[np.ndarray] = []
    print(f"Decoding {T} time bins using {N} neurons…")
    rate_tau = rate_map_hd * tau  # expected spike count per bin
    for t in tqdm(range(T), desc="Decoding time bins"):
        if np.isnan(avg_vel[t]) or avg_vel[t] < vel_threshold:
            posterior_all.append(np.full((H,), np.nan))
            continue
        k = spike_counts[t][None, :]  # (1,N)
        with np.errstate(divide='ignore', invalid='ignore'):
            log_like = np.nansum(k * np.log(rate_tau + eps) - rate_tau, axis=1)  # (H,)
        if np.all(np.isnan(log_like)):
            posterior_all.append(np.full((H,), np.nan))
            continue
        m = np.nanmax(log_like)
        post = np.exp(log_like - m)
        post_sum = np.nansum(post)
        if post_sum <= 0 or not np.isfinite(post_sum):
            posterior_all.append(np.full((H,), np.nan))
            continue
        post /= post_sum
        posterior_all.append(post)
        h_idx = int(np.nanargmax(post))
        decoded_hd[t] = 0.5 * (hd_bins[h_idx] + hd_bins[h_idx + 1])
    return decoded_hd, posterior_all


def xy_bayesian_decoding(
    rate_map_xy: np.ndarray,  # (Y, X, N) mean firing rate (Hz)
    spike_counts: np.ndarray,  # (T, N) counts per bin
    tau: float,
    avg_vel: np.ndarray,  # (T,)
    vel_threshold: float,
    x_edges: np.ndarray,
    y_edges: np.ndarray,
    eps: float = 1e-10,
):
    """2D spatial Bayesian decoding with Poisson likelihood."""
    T, N = spike_counts.shape
    Y, X, N_map = rate_map_xy.shape
    assert N_map == N, "rate_map_xy last dim must match spike_counts."
    decoded_xy = np.full((T, 2), np.nan)
    posterior_all: List[np.ndarray] = []
    print(f"Decoding {T} time bins using {N} neurons over {Y}×{X} grid…")
    rate_tau = rate_map_xy * tau  # expected counts per bin
    rate_tau_flat = rate_tau.reshape(Y * X, N)
    for t in tqdm(range(T), desc="Decoding time bins"):
        if np.isnan(avg_vel[t]) or avg_vel[t] < vel_threshold:
            posterior_all.append(np.full((Y, X), np.nan))
            continue
        k = spike_counts[t]
        with np.errstate(divide='ignore', invalid='ignore'):
            log_like = np.nansum(k[None, :] * np.log(rate_tau_flat + eps) - rate_tau_flat, axis=1)  # (Y*X,)
        if np.all(np.isnan(log_like)):
            posterior_all.append(np.full((Y, X), np.nan))
            continue
        m = np.nanmax(log_like)
        post = np.exp(log_like - m)
        s = np.nansum(post)
        if s <= 0 or not np.isfinite(s):
            posterior_all.append(np.full((Y, X), np.nan))
            continue
        post = (post / s).reshape(Y, X)
        posterior_all.append(post)
        max_idx_flat = int(np.nanargmax(post))
        iy, ix = divmod(max_idx_flat, X)
        x_c = 0.5 * (x_edges[ix] + x_edges[ix + 1])
        y_c = 0.5 * (y_edges[iy] + y_edges[iy + 1])
        decoded_xy[t, 0] = x_c
        decoded_xy[t, 1] = y_c
    return decoded_xy, posterior_all

def main():

    parser.add_argument("--rat", type=str, required=True, help="Rat ID (e.g., 'R1')")

    # Mode
    parser.add_argument("--mode", type=str, choices=["hd", "xy"], default="hd",
                        help="Decode head direction (hd) or location (xy)")

                        help="(Optional) Shuffle HD trajectory before building HD maps")

    # HD params

    # XY params

    parser.add_argument("--n_y_bins", type=int, default=10,
                        help="Number of Y bins (xy mode)")
    parser.add_argument("--arena_pad", type=float, default=0.0,
                        help="Pad inferred arena bounds by this many position units")

    # Timing / thresholds
    parser.add_argument("--tau", type=float, default=0.120,
                        help="Time bin size (s)")
    parser.add_argument("--vel_threshold", type=float, default=5.0,
                        help="Velocity threshold (units/s)")

    # CV + filtering + smoothing
    parser.add_argument("--cv_folds", type=int, default=None,
                        help="Number of CV folds; 0 = no CV. If omitted, you will be prompted.")
    parser.add_argument("--filter_vel", action="store_true", default=True,
                        help="Filter low-velocity bins")
    parser.add_argument("--seed", type=int, default=777, help="Random seed")
    parser.add_argument("--smooth_sigma", type=float, default=1.0,
                        help="Sigma for Gaussian smoothing of HD curves (in bins; circular)")
    parser.add_argument("--save_intermediates", action="store_true", default=False,
                        help="If set, save intermediate artifacts (rate maps, masks, etc.)")

    args = parser.parse_args()

    if args.mode == "hd":
        args.filter_vel = False
        args.vel_threshold = 0.0

    # Resolve cv_folds, optionally prompt
    if args.cv_folds is None:
        try:
            user_in = input("How many CV folds? (0 = no CV): ").strip()
            args.cv_folds = int(user_in)
        except Exception:
            print("No valid input; defaulting to 0 (no CV).")
            args.cv_folds = 0
    if args.cv_folds < 0:
        raise ValueError("cv_folds must be >= 0")

    data_path = Path(args.data_path)
    save_root = Path(args.save_path) / f"Dataset_{args.dataset}" / args.rat
    os.makedirs(save_root, exist_ok=True)

    print(f"\n=== {'Head-Direction' if args.mode=='hd' else 'Location (XY)'} Decoding Pipeline (Dusz) ===")
    print(f"Data path : {data_path}")
    print(f"Save path : {save_root}")
    if args.mode == "hd":
        print(f"HD bins : {args.n_hd_bins}")
        if args.shuffle_hd:
            print("Shuffling HD (pre-map)")
    else:
        print(f"XY bins : {args.n_y_bins} × {args.n_x_bins}")
    print(f"tau: {args.tau:.3f}s | vel_th: {args.vel_threshold} | CV folds: {args.cv_folds}")

    t0 = time.time()
    print("\nLoading data…")
    spike_times_raw, positions, cell_types, velocity_series = load_dusz_data(data_path, args.dataset, args.rat)
    excit_ids = extract_excitatory_ids(cell_types)

        f"Spikes: {len(spike_times_raw)} | Excitatory neurons: {len(excit_ids)} | Position samples: {positions.shape[0]}"

    if spike_times_raw.size == 0 or positions.size == 0 or excit_ids.size == 0:

    base_time = np.arange(spike_times_raw[0, 0], spike_times_raw[-1, 0] + args.tau, args.tau)
    print(
        f"Time bins: {len(base_time)-1} bins covering {base_time[0]:.3f}–{base_time[-1]:.3f}s"

    print("Binning spikes into rates…")
    spike_rates = build_spike_rates(spike_times_raw, base_time, args.tau, excit_ids)  # (T, N)

    # If velocity_series not provided, derive from positions per-sample

            print(f"Total time bins: {len(decoded_hd)}")

                velocity_threshold=args.vel_threshold,
                filter_vel=args.filter_vel,
                arena_pad=args.arena_pad,
            )
            print("Using ALL-DATA XY tuning; decoding all bins.")
            spike_counts = (spike_rates * args.tau).astype(float)
            avg_vel = compute_velocity(positions, base_time)
            time_centers = 0.5 * (base_time[:-1] + base_time[1:])
            if args.smooth_sigma and args.smooth_sigma > 0:
                rate_map_xy = _gaussian_filter_xy(rate_map_xy, (args.smooth_sigma, args.smooth_sigma))
            if args.save_intermediates:
                np.save(save_root / f"xy_rate_map_{args.n_y_bins}x{args.n_x_bins}.npy", rate_map_xy)
                np.save(save_root / f"xy_occupancy_map_{args.n_y_bins}x{args.n_x_bins}.npy", occupancy_xy)

            decoded_xy, posterior_all = xy_bayesian_decoding(

                )
                print("Saving decoding outputs…")
                base = f"decoded_xy_cv{cv_idx}_{args.n_y_bins}x{args.n_x_bins}"
                np.save(save_root / f"{base}.npy", decoded_xy)
                np.save(save_root / f"{base}_posterior.npy", np.array(posterior_all, dtype=object))
                np.save(save_root / f"{base}_time_centers.npy", time_centers)

                eval_res = evaluate_errors_xy(decoded_xy, time_centers, positions, velocity_series, args.vel_threshold)
                np.savez_compressed(save_root / f"errors_xy_cv{cv_idx}_{args.n_y_bins}x{args.n_x_bins}.npz", **eval_res)
                print("Saved error analysis:", save_root / f"errors_xy_cv{cv_idx}_{args.n_y_bins}x{args.n_x_bins}.npz")

                fold_record: Dict[str, Any] = {
                    "cv_idx": int(cv_idx),
                    "n_test_bins": int(len(test_idx)),
                    "output_prefix": base,
                    "error_stats": eval_res["stats"],
                }
                cv_summary["folds"].append(fold_record)

            with open(save_root / f"cv_summary_xy_{args.n_y_bins}x{args.n_x_bins}.json", "w") as f:
                json.dump(cv_summary, f, indent=2)
            print("\nWrote CV summary:", save_root / f"cv_summary_xy_{args.n_y_bins}x{args.n_x_bins}.json")

    print(f"\nDone in {(time.time()-t0)/60.0:.2f} min.")

    main()
