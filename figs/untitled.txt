import numpy as np

def rotated_tuning(s1, s2, c, theta, sigma, amp, wrap=False):
    if wrap:
        d1 = ((s1 - c + 0.5) % 1.0) - 0.5 
        d2 = ((s2 - c + 0.5) % 1.0) - 0.5
    else:
        d1 = s1 - c
        d2 = s2 - c
        
    ct, st = np.cos(theta), np.sin(theta)
    u = d1 * ct - d2 * st
    
    return amp * np.exp(-0.5 * (u / sigma) ** 2)

    use_wrap = False
    
    beta0     = np.log(min_rate)
    amplitude = np.log(max_rate) - beta0
    theta = np.deg2rad(theta_deg)
    
    rng = np.random.default_rng(seed)
    s1 = rng.uniform(lo, hi, size=N)
    s2 = rng.uniform(lo, hi, size=N)
    
    bin_edges = np.linspace(lo, hi, num_bins + 1)
    
    tuning = rotated_tuning(s1, s2, c=mu, theta=theta, sigma=sigma, amp=amplitude, wrap=use_wrap)
    
    rates  = np.exp(beta0 + tuning)
    spike_counts = rng.poisson(rates)
    return spike_counts, (s1, s2)

import numpy as np
from scipy.ndimage import gaussian_filter

def nan_aware_gaussian_filter(rate_map: np.ndarray, sigma: int) -> np.ndarray:
    V = rate_map.copy()
    isnan = np.isnan(V)
    V[isnan] = 0.0
    U = np.ones_like(rate_map, dtype=float)
    U[isnan] = 0.0

    VV = gaussian_filter(V, sigma=sigma, mode="nearest")
    UU = gaussian_filter(U, sigma=sigma, mode="nearest")

    out = VV / UU
    out[UU <= 0] = np.nan
    return out

def get_splits(
        visits, # s1_dim, s2_dim, n_max_visits, n_neurons
        min_visits=2,
        cap=False,
        seed=None
):
    """
    Randomly halves the samples in each bin → train / test splits.

    Returns
    -------
    train_mean, test_mean : mean spike count in each bin (nan if <min_visits)
    train_occ,  test_occ  : visit counts assigned to each split
    """
    rng = np.random.default_rng(seed)
    num_bins = visits.shape[0]

    # half = cap // 2
    n_neurons = visits.shape[3]

    total_mean = np.full((num_bins, num_bins, n_neurons), np.nan)
    train_mean = np.full((num_bins, num_bins, n_neurons), np.nan)
    test_mean  = np.full((num_bins, num_bins, n_neurons), np.nan)
    train_occ  = np.full((num_bins, num_bins), 0)
    test_occ   = np.full((num_bins, num_bins), 0)

    for s1 in range(visits.shape[0]):
        for s2 in range(visits.shape[1]):
            for n in range(visits.shape[3]):
                n_visits = np.count_nonzero(~np.isnan(visits[s1, s2, :, n]))
                if n_visits >= min_visits:
                    take_n = min_visits if cap else n_visits
                    visit_split = visits[s1, s2, :take_n, n]#[s1, s2, :cap, n]
                    rng.shuffle(visit_split)
                    half = len(visit_split) // 2
                    train, test = visit_split[:half], visit_split[half:]
                    
                    train_mean[s1, s2, n] = train.mean()
                    test_mean[s1,  s2, n] = test.mean()
                    total_mean[s1, s2, n] = visit_split.mean()
                    train_occ[s1, s2] = half
                    test_occ[s1, s2] = half

    return total_mean, train_mean, test_mean, train_occ, test_occ

def quantify_variance(heatmap_test, heatmap_train, axis=0, verbose=False, normalize=False):
    """
    Quantify the variance due to a certain variable in a cross-validated manner.
    heatmaps are 2D, and are average spikes/sec across visits to each behavioral bin.
    Negative values indicate dominance of inherent variance over dimension-explained variance.
    """

    test_mean = np.nanmean(heatmap_test, axis=axis, keepdims=True)
    train_mean = np.nanmean(heatmap_train, axis=axis, keepdims=True)

    n_bins = heatmap_test.shape[axis]

    # Now compute the variance terms with occupancy weighting
    inter_var_test = np.nansum((heatmap_test - test_mean)**2)
    inter_var_train = np.nansum((heatmap_train - train_mean)**2)

    intra_var_all = np.nansum((heatmap_test - heatmap_train)**2)
    intra_var_means = n_bins * np.nansum((test_mean - train_mean)**2)

    if verbose:
        print(f"axis={axis} sum((test - test_mean)^2)={inter_var_test:.4f}")
        print(f"axis={axis} sum((train - train_mean)^2)={inter_var_train:.4f}")
        print(f"axis={axis} sum((test - train)^2)={intra_var_all:.4f}")
        print(f"axis={axis} n_bins * sum((test_mean - train_mean)^2)={intra_var_means:.4f}")
        print()
    
    estimator = 0.5 * (
        inter_var_test +
        inter_var_train -
        intra_var_all +
        intra_var_means
    )

    # this is a new part of the script and I'm not sure how I feel about it
    # I kind of wonder if we should be normalizing with something like
    # 0.5 * np.nansum((heatmap_test - heatmap_train))**2
    if normalize:
        ev_den = 0.5 * (inter_var_test + inter_var_train)
        estimator = estimator / ev_den if ev_den > 0 else np.nan
    
    return estimator, inter_var_test, inter_var_train, intra_var_all, intra_var_means

def build_raw_visits(s1_t, s2_t, counts, tau, num_bins, bin_edges):
    '''
    Get a list of all of the time bin visits and the associated spike rates for
    each spatial bin -- "raw" rate maps.
    '''
    visit_rates = counts.astype(float) / float(tau)

    i = np.digitize(s1_t, bin_edges) - 1
    j = np.digitize(s2_t, bin_edges) - 1
    in_range = (i >= 0) & (i < num_bins) & (j >= 0) & (j < num_bins)
    i = i[in_range]
    j = j[in_range]
    visit_rates = visit_rates[in_range]

    occ_visits, _, _ = np.histogram2d(
        s1_t[in_range], s2_t[in_range], bins=[bin_edges, bin_edges]
    )
    occ_visits = occ_visits.astype(int)

    max_bin_occ = int(occ_visits.max()) if occ_visits.size else 0

    bin_lists = [[[] for _ in range(num_bins)] for _ in range(num_bins)]
    for idx_visit, (ii, jj) in enumerate(zip(i, j)):
        bin_lists[ii][jj].append(idx_visit)

    raw_visits = np.full((num_bins, num_bins, max_bin_occ), np.nan)
    
    for ii in range(num_bins):
        for jj in range(num_bins):
            idxs = bin_lists[ii][jj]
            if not idxs:
                continue

            take = len(visit_rates[idxs])
            raw_visits[ii, jj, :take] = visit_rates[idxs][:take]

    return raw_visits, occ_visits

def ground_truth_evs(theta_deg, *, lo=0.0, hi=1.0, mu=0.4, sigma=0.1,
                     min_rate=5.0, max_rate=35.0, use_wrap=False, grid=256):
    beta0     = np.log(min_rate)
    amplitude = np.log(max_rate) - beta0
    theta = np.deg2rad(theta_deg)

    s1_vals = np.linspace(lo, hi, grid)
    s2_vals = np.linspace(lo, hi, grid)
    S1, S2 = np.meshgrid(s1_vals, s2_vals, indexing='ij')

    tuning = rotated_tuning(S1, S2, c=mu, theta=theta, sigma=sigma, amp=amplitude, wrap=use_wrap)
    lam = np.exp(beta0 + tuning)                 # λ(s1,s2)

    # expectations over the uniform grid
    lam_mean_s2 = lam.mean(axis=1)               # E[λ | s1]
    lam_mean_s1 = lam.mean(axis=0)               # E[λ | s2]
    var_total   = lam.var()
    ev_x = lam_mean_s2.var() / var_total
    ev_y = lam_mean_s1.var() / var_total
    return float(ev_x), float(ev_y)
